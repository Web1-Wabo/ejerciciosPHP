        $text = [];
        $texto = strtolowe($texto); 
        foreach($catalogo as $libro){
            if($libro["autor"] === $texto || $libro["titulo"] === $texto){ -> aqui convertir a miniscula
                //strtolower($libro["autor"]) === $texto, '''''
                $text[] = $libro;
            }
        }

        return $text;
    }
